cmake_minimum_required(VERSION 3.16)
project(FreshCart VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FRESHCART_BUILD_GUI "Build the Qt desktop application" ON)
option(FRESHCART_GUI_TESTS "Build offscreen GUI integration checks" OFF)

add_library(freshcart_core src/store.cpp)
target_include_directories(freshcart_core PUBLIC src)
if(MSVC)
    target_compile_options(freshcart_core PRIVATE /W4 /utf-8)
else()
    target_compile_options(freshcart_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(FreshCartConsole src/console.cpp)
target_link_libraries(FreshCartConsole PRIVATE freshcart_core)

include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE freshcart_core)
    add_test(NAME billing_and_inventory COMMAND core_tests)
endif()

if(FRESHCART_BUILD_GUI)
    list(APPEND CMAKE_PREFIX_PATH "C:/Qt/6.12.1/msvc2022_64")
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
    add_executable(FreshCart WIN32 src/main.cpp src/mainwindow.cpp src/mainwindow.h)
    target_link_libraries(FreshCart PRIVATE freshcart_core Qt${QT_VERSION_MAJOR}::Widgets)
    if(MSVC)
        target_compile_options(FreshCart PRIVATE /W4 /utf-8)
    endif()
    if(FRESHCART_GUI_TESTS)
        add_executable(gui_smoke tests/gui_smoke.cpp src/mainwindow.cpp)
        target_link_libraries(gui_smoke PRIVATE freshcart_core Qt${QT_VERSION_MAJOR}::Widgets)
        add_test(NAME gui_workflow COMMAND gui_smoke ${CMAKE_CURRENT_BINARY_DIR}/screenshots)
        set_tests_properties(gui_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endif()
